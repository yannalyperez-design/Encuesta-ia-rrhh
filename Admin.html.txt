    <a href="?page=admin">Panel admin</a>
    <span class="tag"><?= email ?></span>
  </div>
  <div class="card">
    <div>
      <button class="tabbtn active" id="tab-analitica" onclick="showTab('analitica')">Analítica</button>
      <button class="tabbtn" id="tab-respuestas" onclick="showTab('respuestas')">Respuestas</button>
      <button class="tabbtn" id="tab-accesos" onclick="showTab('accesos')">Accesos</button>
    </div>
    <div id="msg"></div>
    <div id="panel-analitica"></div>
    <div id="panel-respuestas" style="display:none;"></div>
    <div id="panel-accesos" style="display:none;"></div>
  </div>
</div>
<script>
  showTab('analitica');
  function showTab(t){
    ['analitica','respuestas','accesos'].forEach(function(x){
      document.getElementById('tab-'+x).classList.toggle('active', x===t);
      document.getElementById('panel-'+x).style.display = x===t ? 'block' : 'none';
    });
    if (t==='analitica') cargarAnalitica();
    if (t==='respuestas') cargarRespuestas();
    if (t==='accesos') cargarAccesos();
  }

  function cargarAnalitica(){
    document.getElementById('panel-analitica').innerHTML = 'Cargando...';
    google.script.run.withSuccessHandler(function(data){
      var html = '<div class="stat">' + data.total + '</div><p class="sub">respuestas registradas</p>';
      html += '<button onclick="generarInsights()">Generar insights con IA</button><div id="insights" style="margin-top:14px;"></div>';
      Object.keys(data.preguntas).forEach(function(id){
        var q = data.preguntas[id];
        html += '<h2 style="margin-top:22px;">' + q.texto + '</h2>';
        var total = Object.values(q.counts).reduce(function(a,b){return a+b;},0) || 1;
        Object.keys(q.counts).forEach(function(op){
          var pct = Math.round((q.counts[op]/total)*100);
          html += '<div class="bar-row"><div class="name">' + op + '</div><div class="bar-track"><div class="bar-fill" style="width:' + pct + '%"></div></div><div class="pct">' + pct + '%</div></div>';
        });
      });
      document.getElementById('panel-analitica').innerHTML = html;
    }).withFailureHandler(function(e){ document.getElementById('panel-analitica').innerHTML = '<div class="msg err">' + e.message + '</div>'; }).getAnaliticaAdmin();
  }

  function generarInsights(){
    document.getElementById('insights').innerHTML = 'Generando...';
    google.script.run.withSuccessHandler(function(r){
      document.getElementById('insights').innerHTML = r.ok ? '<div class="card" style="padding:16px;white-space:pre-wrap;">' + r.texto + '</div>' : '<div class="msg err">' + r.error + '</div>';
    }).getInsightsIA();
  }

  function cargarRespuestas(){
    document.getElementById('panel-respuestas').innerHTML = 'Cargando...';
    google.script.run.withSuccessHandler(function(rows){
      var html = '<table><tr><th>Fecha</th><th>Nombre</th><th>Área</th><th>Cédula</th><th>Correo</th></tr>';
      rows.forEach(function(r){
        html += '<tr><td>' + new Date(r.timestamp).toLocaleString() + '</td><td>' + r.nombre + ' ' + r.apellido + '</td><td>' + r.area + '</td><td>' + r.cedula + '</td><td>' + r.email + '</td></tr>';
      });
      html += '</table>';
      document.getElementById('panel-respuestas').innerHTML = html || 'Sin respuestas todavía.';
    }).withFailureHandler(function(e){ document.getElementById('panel-respuestas').innerHTML = '<div class="msg err">' + e.message + '</div>'; }).getRespuestasAdmin();
  }

  function cargarAccesos(){
    document.getElementById('panel-accesos').innerHTML = 'Cargando...';
    google.script.run.withSuccessHandler(function(rows){
      var html = '<table><tr><th>Correo</th><th>Rol</th><th></th></tr>';
      rows.forEach(function(r){
        html += '<tr><td>' + r.email + '</td><td>' + r.rol + '</td><td><button class="danger" onclick="quitar(\'' + r.email + '\')">Quitar</button></td></tr>';
      });
      html += '</table>';
      html += '<div style="margin-top:16px;"><label>Correo</label><input type="email" id="nuevoEmail">' +
        '<label>Rol</label><select id="nuevoRol"><option value="usuario">Usuario</option><option value="instructor">Instructor</option><option value="admin">Admin</option></select>' +
        '<button onclick="agregar()">Agregar / actualizar</button></div>';
      document.getElementById('panel-accesos').innerHTML = html;
    }).getAccesos();
  }
  function agregar(){
    var email = document.getElementById('nuevoEmail').value.trim();
    var rol = document.getElementById('nuevoRol').value;
    if (!email) return;
    google.script.run.withSuccessHandler(function(){ cargarAccesos(); }).guardarAcceso(email, rol);
  }
  function quitar(email){
    if (!confirm('¿Quitar acceso a ' + email + '?')) return;
    google.script.run.withSuccessHandler(function(){ cargarAccesos(); }).eliminarAcceso(email);
  }
</script>
</body></html>
